```
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.28;
contract PerfectNumber {
    function checkPerfectNumber(uint256 n) public pure returns (bool) {
        uint256 sum = 0;
        for (uint256 i = 1; i <= n / 2; i++) {
            if (n % i == 0) {
                sum += i;
            }
        }
        return n > 1 && sum == n;
    }
}
```
Explanation of the code:
pragma solidity ^0.8.28 specifies the compatible Solidity compiler version.
contract PerfectNumber defines the smart contract.
checkPerfectNumber(uint256 n) accepts a positive integer.
sum stores the sum of the proper divisors.
The for loop checks numbers from 1 to n/2.
n % i == 0 checks whether i divides n completely.
If divisible, i is added to sum.
Finally, sum == n checks whether the number is perfect.
The function returns true for a perfect number and false otherwise.
pure indicates that the function does not read or modify blockchain state.

B. main.js
This file connects the frontend to MetaMask and the deployed smart contract.

```
let contract;
let signer;
const contractAddress = "PASTE_DEPLOYED_CONTRACT_ADDRESS";
const abi = [
    "function checkPerfectNumber(uint256 n) public pure returns (bool)"
];
async function connectWallet() {
    if (!window.ethereum) {
        alert("Please install MetaMask!");
        return;
    }
    try {
        const provider = new ethers.BrowserProvider(window.ethereum);
        await provider.send("eth_requestAccounts", []);
        signer = await provider.getSigner();
        contract = new ethers.Contract(
            contractAddress,
            abi,
            signer
        );
        document.getElementById("wallet").innerText =
            "Connected: " + await signer.getAddress();
    } catch (error) {
        alert(error.message);
    }
}
async function checkNumber() {
    try {
        if (!contract) {
            alert("Please connect your wallet first!");
            return;
        }
        const n = document.getElementById("number").value;
        if (
            n === "" ||
            !Number.isSafeInteger(Number(n)) ||
            Number(n) < 1
        ) {
            alert("Enter a valid positive integer!");
            return;
        }
        const result = await contract.checkPerfectNumber(n);
        document.getElementById("result").innerText =
            result
                ? "It is a Perfect Number!"
                : "It is NOT a Perfect Number!";
    } catch (error) {
        document.getElementById("result").innerText =
            "Error: " + error.message;
    }
}
```
Explanation:
contract stores the smart contract instance.
signer represents the connected wallet.
contractAddress stores the deployed contract address.
abi defines the function interface.
connectWallet() connects MetaMask to the frontend.
BrowserProvider connects Ethers.js with MetaMask.
getSigner() obtains the connected wallet account.
ethers.Contract() creates the contract instance.
checkNumber() accepts the user input.
checkPerfectNumber(n) calls the Solidity function.
The returned Boolean value is displayed on the webpage.
Important: After deployment, replace PASTE_DEPLOYED_CONTRACT_ADDRESS with your actual contract address.

// SPDX-License-Identifier: MIT
```
pragma solidity ^0.8.28;
contract Prime {
    function isPrime(uint n) public pure returns(bool) {
        if (n < 2) {
            return false;
        }
        for (uint i = 2; i <= n / i; i++) {
            if (n % i == 0) {
                return false;
            }
        }
        return true;
    }
}
```
Explanation:
contract Prime: Creates a smart contract named Prime.
isPrime(uint n): Accepts one unsigned integer.
public: Allows the function to be called externally.
pure: Does not read or modify blockchain state.
n < 2: Numbers less than 2 are not prime.
for loop: Checks whether the number is divisible by any number starting from 2.
n % i == 0: Checks whether the number is completely divisible by i.
return false: The number is not prime.
return true: The number is prime.
A prime number is a number greater than 1 that is divisible only by 1 and itself.
4. Create the Deployment Module
Inside ignition/modules, create a file named Prime.ts.
Write the following code:
import { buildModule } from "@nomicfoundation/hardhat-ignition/modules";
```
const PrimeModule = buildModule("PrimeModule", (m) => {
    const prime = m.contract("Prime");
    return { prime };
});
export default PrimeModule;
```
Explanation:
buildModule: Creates a deployment module.
m.contract("Prime"): Deploys the Prime smart contract.
return: Returns the deployed contract.
export default: Exports the module.

File 2: main.js
Write the following JavaScript code:
const contractAddress = "PASTE_YOUR_CONTRACT_ADDRESS";
```
const abi = [
    "function isPrime(uint256 n) public pure returns(bool)"
];
let contract;
async function connectWallet() {
    await window.ethereum.request({
        method: "eth_requestAccounts"
    });
    const provider = new ethers.BrowserProvider(window.ethereum);
    const signer = await provider.getSigner();
    contract = new ethers.Contract(
        contractAddress,
        abi,
        signer
    );
    alert("Wallet Connected!");
}
async function checkPrime() {
    const n = document.getElementById("num").value;
    if (n === "") {
        alert("Enter a number");
        return;
    }
    const result = await contract.isPrime(n);
    if (result) {
        document.getElementById("result").innerHTML =
            n + " is a Prime Number";
    }
    else {
        document.getElementById("result").innerHTML =
            n + " is Not a Prime Number";
    }
}
```
Explanation:
contractAddress: Stores the deployed smart contract address.
ABI: Defines the smart contract function.
connectWallet(): Connects MetaMask to the DApp.
BrowserProvider: Connects the application to the blockchain.
getSigner(): Gets the connected wallet account.
ethers.Contract(): Connects the frontend with the deployed contract.
checkPrime(): Takes the number entered by the user.
contract.isPrime(n): Calls the Solidity function.
if (result): Displays Prime Number if the result is true.
else: Displays Not a Prime Number if the result is false.
Important: Replace PASTE_YOUR_CONTRACT_ADDRESS with the actual address obtained during deployment. Keep the double quotation marks.

// SPDX-License-Identifier: MIT
```
pragma solidity ^0.8.28;
contract Voting {
    string[] public candidates;
    mapping(string => uint) public votes;
    mapping(address => bool) public hasVoted;
    constructor() {
        candidates.push("Alice");
        candidates.push("Bob");
        candidates.push("Charlie");
    }
    function vote(string memory candidate) public {
        require(!hasVoted[msg.sender], "Already voted");
        bool found = false;
        for (uint i = 0; i < candidates.length; i++) {
            if (keccak256(bytes(candidates[i])) == keccak256(bytes(candidate))) {
                found = true;
            }
        }
        require(found, "Invalid candidate");
        votes[candidate]++;
        hasVoted[msg.sender] = true;
    }
    function getVotes(string memory candidate) public view returns (uint) {
        return votes[candidate];
    }
    function getCandidates() public view returns (string[] memory) {
        return candidates;
    }
}
```

Step 7: Write script.js
Open:
code frontend\script.js
Paste:

```
let provider;
let signer;
let contract;
const contractAddress = "PASTE_CONTRACT_ADDRESS_HERE";
async function connectWallet() {
    if (!window.ethereum) {
        alert("Please install MetaMask!");
        return;
    }
    try {
        provider = new ethers.BrowserProvider(window.ethereum);
        await provider.send("eth_requestAccounts", []);
        signer = await provider.getSigner();
        const address = await signer.getAddress();
        document.getElementById("account").innerText =
            "Connected: " + address;
        const response = await fetch("abi.json");
        const artifact = await response.json();
        contract = new ethers.Contract(
            contractAddress,
            artifact.abi,
            signer
        );
        document.getElementById("status").innerText =
            "Wallet connected successfully!";
        loadCandidates();
    } catch (error) {
        document.getElementById("status").innerText =
            error.message;
    }
}
async function loadCandidates() {
    if (!contract) return;
    try {
        const candidates = await contract.getCandidates();
        const container = document.getElementById("candidates");
        container.innerHTML = "";
        for (let candidate of candidates) {
            const voteCount = await contract.getVotes(candidate);
            const div = document.createElement("div");
            div.className = "candidate";
            const heading = document.createElement("h3");
            heading.innerText = candidate;
            const votes = document.createElement("p");
            votes.innerText = "Votes: " + voteCount.toString();
            const button = document.createElement("button");
            button.innerText = "Vote";
            button.onclick = function () {
                castVote(candidate);
            };
            div.appendChild(heading);
            div.appendChild(votes);
            div.appendChild(button);
            container.appendChild(div);
        }
    } catch (error) {
        document.getElementById("status").innerText =
            error.message;
    }
}
async function castVote(candidate) {
    try {
        const transaction = await contract.vote(candidate);
        document.getElementById("status").innerText =
            "Waiting for confirmation...";
        await transaction.wait();
        document.getElementById("status").innerText =
            "Vote successfully submitted for " + candidate;
        loadCandidates();
    } catch (error) {
        document.getElementById("status").innerText =
            error.reason || error.shortMessage || error.message;
    }
}
```
